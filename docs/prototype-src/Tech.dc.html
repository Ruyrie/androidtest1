Glide 缓存配置
@GlideModule
public final class HeBanGlideModule
        extends AppGlideModule {
  @Override
  public void applyOptions(Context ctx,
                           GlideBuilder b) {
    // 内存 20MB，磁盘 250MB
    b.setMemoryCache(
        new LruResourceCache(20L * 1024 * 1024));
    b.setDiskCache(
        new InternalCacheDiskCacheFactory(
            ctx, 250L * 1024 * 1024));
    b.setDefaultRequestOptions(
        new RequestOptions()
            .diskCacheStrategy(
                DiskCacheStrategy.AUTOMATIC)
            .placeholder(R.drawable.ph_image));
  }
}
